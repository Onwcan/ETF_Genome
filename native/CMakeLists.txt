cmake_minimum_required(VERSION 3.16)
project(etf_genome_native VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(ETF_GENOME_NETWORK_DEFAULT ON)
else()
    set(ETF_GENOME_NETWORK_DEFAULT OFF)
endif()
option(ETF_GENOME_BUILD_NETWORK "Build the Linux epoll synthetic feed" ${ETF_GENOME_NETWORK_DEFAULT})
if(ETF_GENOME_BUILD_NETWORK AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native network layer currently requires Linux")
endif()

option(ETF_GENOME_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(ETF_GENOME_ENABLE_TSAN "Enable ThreadSanitizer (separate build)" OFF)
option(ETF_GENOME_USE_HARDWARE_INTERFERENCE_SIZE "Use compiler-specific cache line padding (ABI sensitive)" OFF)
option(ETF_GENOME_WARNINGS_AS_ERRORS "Treat project compiler warnings as errors" OFF)
if(ETF_GENOME_ENABLE_SANITIZERS AND ETF_GENOME_ENABLE_TSAN)
    message(FATAL_ERROR "Use separate builds for ASan/UBSan and TSan")
endif()

find_package(Threads REQUIRED)
add_library(etf_genome_native
    src/exposure.cpp src/csv.cpp src/replay.cpp src/price_exposure.cpp src/snapshot.cpp
    src/protocol.cpp src/decoder.cpp)
target_include_directories(etf_genome_native PUBLIC include)
target_compile_features(etf_genome_native PUBLIC cxx_std_20)
set_target_properties(etf_genome_native PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(etf_genome_native PUBLIC Threads::Threads)
if(ETF_GENOME_USE_HARDWARE_INTERFERENCE_SIZE)
    target_compile_definitions(etf_genome_native PUBLIC ETF_GENOME_USE_HARDWARE_INTERFERENCE_SIZE=1)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(etf_genome_native PUBLIC ETF_GENOME_PLATFORM_LINUX=1)
endif()
if(ETF_GENOME_BUILD_NETWORK)
    target_sources(etf_genome_native PRIVATE src/socket.cpp src/epoll.cpp src/feed.cpp)
    target_compile_definitions(etf_genome_native PUBLIC ETF_GENOME_HAS_NETWORK=1)
    add_executable(etf-genome-feed-server apps/feed_server.cpp)
    add_executable(etf-genome-feed-client apps/feed_client.cpp)
    target_link_libraries(etf-genome-feed-server PRIVATE etf_genome_native)
    target_link_libraries(etf-genome-feed-client PRIVATE etf_genome_native)
elseif(WIN32)
    target_compile_definitions(etf_genome_native PUBLIC ETF_GENOME_PLATFORM_WINDOWS=1)
endif()

if(MSVC)
    if(ETF_GENOME_ENABLE_SANITIZERS OR ETF_GENOME_ENABLE_TSAN)
        message(FATAL_ERROR "These sanitizer options require a supported GCC/Clang environment")
    endif()
    target_compile_options(etf_genome_native PUBLIC /W4 /permissive-)
    if(ETF_GENOME_WARNINGS_AS_ERRORS)
        target_compile_options(etf_genome_native PUBLIC /WX)
    endif()
else()
    target_compile_options(etf_genome_native PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(ETF_GENOME_WARNINGS_AS_ERRORS)
        target_compile_options(etf_genome_native PUBLIC -Werror)
    endif()
    if(ETF_GENOME_ENABLE_SANITIZERS)
        target_compile_options(etf_genome_native PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(etf_genome_native PUBLIC -fsanitize=address,undefined)
    elseif(ETF_GENOME_ENABLE_TSAN)
        target_compile_options(etf_genome_native PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(etf_genome_native PUBLIC -fsanitize=thread)
    endif()
endif()

add_executable(etf-genome-native src/main.cpp)
target_link_libraries(etf-genome-native PRIVATE etf_genome_native)
add_executable(etf-genome-benchmark apps/benchmark.cpp benchmarks/allocation_counter.cpp)
target_include_directories(etf-genome-benchmark PRIVATE benchmarks)
target_link_libraries(etf-genome-benchmark PRIVATE etf_genome_native)
set(ETF_GENOME_CONFIG_FLAGS
    "$<$<CONFIG:Debug>:${CMAKE_CXX_FLAGS_DEBUG}>$<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}>$<$<CONFIG:RelWithDebInfo>:${CMAKE_CXX_FLAGS_RELWITHDEBINFO}>$<$<CONFIG:MinSizeRel>:${CMAKE_CXX_FLAGS_MINSIZEREL}>")
if(MSVC)
    set(ETF_GENOME_PROJECT_FLAGS "/W4 /permissive- /std:c++20")
    if(ETF_GENOME_WARNINGS_AS_ERRORS)
        string(APPEND ETF_GENOME_PROJECT_FLAGS " /WX")
    endif()
else()
    set(ETF_GENOME_PROJECT_FLAGS "-Wall -Wextra -Wpedantic -Wconversion -Wshadow -std=c++20")
    if(ETF_GENOME_WARNINGS_AS_ERRORS)
        string(APPEND ETF_GENOME_PROJECT_FLAGS " -Werror")
    endif()
    if(ETF_GENOME_ENABLE_SANITIZERS)
        string(APPEND ETF_GENOME_PROJECT_FLAGS " -fsanitize=address,undefined -fno-omit-frame-pointer")
    elseif(ETF_GENOME_ENABLE_TSAN)
        string(APPEND ETF_GENOME_PROJECT_FLAGS " -fsanitize=thread -fno-omit-frame-pointer")
    endif()
endif()
target_compile_definitions(etf-genome-benchmark PRIVATE
    ETF_GENOME_COMPILER_ID="${CMAKE_CXX_COMPILER_ID}"
    ETF_GENOME_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}"
    ETF_GENOME_BUILD_TYPE="$<CONFIG>"
    ETF_GENOME_OPTIMIZATION_FLAGS="${ETF_GENOME_CONFIG_FLAGS}"
    ETF_GENOME_COMPILE_FLAGS="${CMAKE_CXX_FLAGS} ${ETF_GENOME_CONFIG_FLAGS} ${ETF_GENOME_PROJECT_FLAGS}")

include(CTest)
if(BUILD_TESTING)
    add_executable(native-tests tests/test_native.cpp)
    target_link_libraries(native-tests PRIVATE etf_genome_native)
    add_test(NAME native-unit COMMAND native-tests)
    add_executable(protocol-tests tests/test_protocol.cpp)
    target_link_libraries(protocol-tests PRIVATE etf_genome_native)
    add_test(NAME native-protocol COMMAND protocol-tests)
    add_executable(exposure-stream-tests tests/test_stream_exposure.cpp)
    target_link_libraries(exposure-stream-tests PRIVATE etf_genome_native)
    add_test(NAME native-stream-exposure COMMAND exposure-stream-tests)
    add_test(NAME native-benchmark-smoke COMMAND etf-genome-benchmark --mode decoder --events 1000 --allocations)
    set_tests_properties(native-benchmark-smoke PROPERTIES PASS_REGULAR_EXPRESSION "operator_new_calls\":0")
    add_executable(allocation-counter-tests benchmarks/test_allocation_counter.cpp benchmarks/allocation_counter.cpp)
    target_include_directories(allocation-counter-tests PRIVATE benchmarks)
    target_link_libraries(allocation-counter-tests PRIVATE etf_genome_native)
    add_test(NAME native-allocation-counter COMMAND allocation-counter-tests)
    if(ETF_GENOME_BUILD_NETWORK)
        add_executable(network-tests tests/test_network.cpp)
        target_link_libraries(network-tests PRIVATE etf_genome_native)
        add_test(NAME native-network COMMAND network-tests)
        set_tests_properties(native-network PROPERTIES TIMEOUT 60)
    endif()
    add_test(NAME native-csv-smoke COMMAND etf-genome-native shock
        --edges ${CMAKE_CURRENT_SOURCE_DIR}/examples/synthetic_edges.csv
        --shocks ${CMAKE_CURRENT_SOURCE_DIR}/examples/synthetic_shocks.csv)
    set_tests_properties(native-csv-smoke PROPERTIES
        PASS_REGULAR_EXPRESSION "synthetic:ETF_A,-0.02")
    add_test(NAME native-replay-smoke COMMAND etf-genome-native replay 10000)
    set_tests_properties(native-replay-smoke PROPERTIES PASS_REGULAR_EXPRESSION "events=10000")
    add_test(NAME native-invalid-count COMMAND etf-genome-native replay -1)
    set_tests_properties(native-invalid-count PROPERTIES WILL_FAIL TRUE)
    set_tests_properties(native-unit native-protocol native-stream-exposure native-csv-smoke native-replay-smoke native-invalid-count
        PROPERTIES TIMEOUT 60)
endif()
